# Декоратор для версионирования функций (из нашего прошлого обсуждения)
# -----------------------------------------------------------------------------
def versioned(version: str):
    """Декоратор, добавляющий атрибут __version__ к функции."""
    def decorator(func):
        func.__version__ = version
        return func
    return decorator

@versioned("1.1.0")  # Пункт 2: Версия конкретной функции
def run_game(package_name: str, activity_name: str) -> Optional[Dict[str, Any]]:
    """
    Запускает указанное Android-приложение через ADB.
    
    Args:
        package_name (str): Имя пакета приложения (напр., 'com.example.game').
        activity_name (str): Имя активности (напр., 'com.example.game.MainActivity').
        
    Returns:
        dict | None: Словарь с результатом выполнения или None при критической ошибке.
            Формат при успехе: {"status": "success", "time_ms": float, "message": str}
            Формат при ошибке: {"status": "error", "time_ms": float, "message": str}
            
    .. versionadded:: 1.0.0
       Базовая реализация запуска через subprocess.
    .. versionchanged:: 1.1.0
       Добавлены замеры времени, логирование и типизированный возврат dict.
    """
    # Пункт 5: Начинаем замер времени (perf_counter точнее, чем time.time)
    start_time = time.perf_counter()
    
    # Пункт 3: Логируем начало действия (уровень INFO)
    logging.info(f"Запуск приложения: {package_name}/{activity_name}")
    
    try:
        result = subprocess.run(
            ["adb", "shell", "am", "start", "-f", "0x10020000", "-n", f"{package_name}/{activity_name}"],
            capture_output=True,
            text=True,
            timeout=10  # ВАЖНО: защита от зависания ADB (максимум 10 секунд)
        )
        
        # Завершаем замер времени и переводим в миллисекунды
        elapsed_time_ms = (time.perf_counter() - start_time) * 1000
        
        # Пункт 4: Возвращаем структурированный результат
        if result.returncode == 0:
            logging.info(f"Успешно запущено за {elapsed_time_ms:.2f} мс")
            return {
                "status": "success",
                "time_ms": round(elapsed_time_ms, 2),
                "message": f"Игра {package_name} запущена"
            }
        else:
            # Очищаем сообщение об ошибке от лишних переносов строк
            error_msg = result.stderr.strip() if result.stderr else "Неизвестная ошибка ADB"
            logging.error(f"Ошибка ADB (код {result.returncode}): {error_msg}")
            return {
                "status": "error",
                "time_ms": round(elapsed_time_ms, 2),
                "message": error_msg
            }

        logging.error("Превышено время ожидания ответа от ADB (timeout)")
        return {"status": "error", "time_ms": 10000.0, "message": "ADB timeout"}

        # logging.exception автоматически добавляет трассировку стека (traceback) в лог

        return None  # Возврат None при полном крахе функции

import time
import logging
from functools import wraps # Важная штука, сохраняет имя и докстринг функции

def timer(func):
    """Декоратор для замера времени выполнения функции."""
    @wraps(func)
    def wrapper(*args, **kwargs):
        # 1. Засекаем время ДО
        start = time.perf_counter()
        
        # 2. Вызываем саму функцию (то, что она должна делать)
        result = func(*args, **kwargs) 
        
        # 3. Засекаем время ПОСЛЕ
        elapsed_ms = (time.perf_counter() - start) * 1000
        
        # 4. Логируем (func.__name__ автоматически подставит имя функции, например 'run_game')
        logging.info(f"⏱ Функция '{func.__name__}' отработала за {elapsed_ms:.2f} мс")
        
        # 5. Возвращаем результат работы функции
        return result
    return wrapper
