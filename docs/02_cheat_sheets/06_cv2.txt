cheatsheet_content = """
================================================================================
              ШПАРГАЛКА CV2 - РАБОТА С ИЗОБРАЖЕНИЯМИ
================================================================================

================================================================================
1. ОТКРЫТИЕ ФАЙЛА
================================================================================

import cv2

# Цветное (BGR) - по умолчанию
img = cv2.imread("photo.png")

# Оттенки серого - сразу при загрузке (быстрее)
img_gray = cv2.imread("photo.png", cv2.IMREAD_GRAYSCALE)

# С альфа-каналом (RGBA)
img_rgba = cv2.imread("photo.png", cv2.IMREAD_UNCHANGED)

# ПРОВЕРКА - если файла нет, imread вернёт None (не ошибку!)
if img is None:
    print("Файл не найден")


================================================================================
2. СОХРАНЕНИЕ ФАЙЛА
================================================================================

# Сохранить в PNG (без потерь)
cv2.imwrite("out.png", img)

# Сохранить в JPEG (с сжатием, качество 0-100)
cv2.imwrite("out.jpg", img, [cv2.IMWRITE_JPEG_QUALITY, 90])

# Сохранить в WebP
cv2.imwrite("out.webp", img, [cv2.IMWRITE_WEBP_QUALITY, 90])


================================================================================
3. ПРЕОБРАЗОВАНИЕ ДЛЯ PIL
================================================================================

from PIL import Image
import numpy as np

# cv2 (BGR) -> PIL (RGB)
pil_img = Image.fromarray(cv2.cvtColor(img, cv2.COLOR_BGR2RGB))

# cv2 gray -> PIL
pil_gray = Image.fromarray(img_gray)

# PIL -> cv2 (RGB -> BGR)
cv2_img = cv2.cvtColor(np.array(pil_img), cv2.COLOR_RGB2BGR)

# PIL gray -> cv2
cv2_gray = np.array(pil_gray)  # тут уже без конвертации цвета


================================================================================
4. ПРЕОБРАЗОВАНИЕ В ЧЁРНО-БЕЛОЕ
================================================================================

# Шаг 1: в оттенки серого (если ещё не серое)
gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

# Вариант А: жёсткий порог (128 - граница)
_, bw = cv2.threshold(gray, 128, 255, cv2.THRESH_BINARY)

# Вариант Б: Otsu - автоподбор порога (лучше для скриншотов)
_, bw = cv2.threshold(gray, 0, 255, cv2.THRESH_BINARY + cv2.THRESH_OTSU)

# Вариант В: адаптивный порог (для неравномерного фона)
bw = cv2.adaptiveThreshold(
    gray, 255,
    cv2.ADAPTIVE_THRESH_GAUSSIAN_C,
    cv2.THRESH_BINARY,
    11, 2  # размер блока, константа
)

# Инверсия (чёрное <-> белое)
bw_inv = cv2.bitwise_not(bw)


================================================================================
5. ПОИСК ФРАГМЕНТА (template matching)
================================================================================

# Шаблон - картинка, которую ищем
template = cv2.imread("fragment.png", cv2.IMREAD_GRAYSCALE)
# Искомое изображение - тоже серое
img_gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)

h_t, w_t = template.shape[:2]  # размеры шаблона

# Поиск - возвращает карту совпадений
result = cv2.matchTemplate(img_gray, template, cv2.TM_CCOEFF_NORMED)

# Методы поиска:
#   TM_CCOEFF_NORMED  - лучший, нормализованный (от -1 до 1)
#   TM_CCORR_NORMED   - корреляция
#   TM_SQDIFF_NORMED  - разница (меньше = лучше)

# Ищем максимум (или минимум для SQDIFF)
min_val, max_val, min_loc, max_loc = cv2.minMaxLoc(result)

# Координаты верхнего левого угла найденного фрагмента
x, y = max_loc

# Рисуем рамку вокруг найденного
cv2.rectangle(img, (x, y), (x + w_t, y + h_t), (0, 255, 0), 2)

# Порог совпадения (0.8 = 80%)
if max_val >= 0.8:
    print(f"Найдено в ({x}, {y}), совпадение: {max_val:.2f}")
else:
    print("Не найдено")


--------------------------------------------------------------------------------
ПОИСК ВСЕХ СОВПАДЕНИЙ (если фрагмент встречается несколько раз)
--------------------------------------------------------------------------------

threshold = 0.8
locations = np.where(result >= threshold)

# zip даёт пары (y, x) - переворачиваем в (x, y)
for pt in zip(*locations[::-1]):
    cv2.rectangle(img, pt, (pt[0] + w_t, pt[1] + h_t), (0, 255, 0), 2)


================================================================================
ГОТОВЫЙ МИНИ-РЕЦЕПТ: НАЙТИ ФРАГМЕНТ НА СКРИНШОТЕ
================================================================================

import cv2

# 1. Загружаем оба изображения в сером
screen = cv2.imread("screenshot.png", cv2.IMREAD_GRAYSCALE)
templ  = cv2.imread("button.png", cv2.IMREAD_GRAYSCALE)

h, w = templ.shape[:2]

# 2. Ищем
res = cv2.matchTemplate(screen, templ, cv2.TM_CCOEFF_NORMED)
_, max_val, _, max_loc = cv2.minMaxLoc(res)

# 3. Результат
if max_val >= 0.85:
    x, y = max_loc
    print(f"Найдено в ({x}, {y}) с точностью {max_val:.3f}")
    # Клик по центру найденного
    click_x = x + w // 2
    click_y = y + h // 2
else:
    print("Не найдено")


================================================================================
"""
