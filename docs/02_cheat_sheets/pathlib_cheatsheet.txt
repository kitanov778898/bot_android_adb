====================================================================
        ШПАРГАЛКА ПО pathlib.Path  (Python 3.4+)
====================================================================

--------------------------------------------------------------------
1. ЗОЛОТЫЕ ПРАВИЛА
--------------------------------------------------------------------
- Свойства (без скобок):  p.parent, p.name
- Методы (со скобками):   p.exists(), p.mkdir()
- Объединение путей:      BASE / "folder" / "file.txt"
- Путь от скрипта:        Path(__file__).parent
- Импорт:                 from pathlib import Path

--------------------------------------------------------------------
2. СВОЙСТВА (без скобок) — "Что это?"
--------------------------------------------------------------------
.parent     Родительская папка
.name       Имя файла с расширением (photo.png)
.stem       Имя без расширения (photo)
.suffix     Расширение (.png)
.suffixes   Список всех расширений (['.tar', '.gz'])
.anchor     Корень пути (/ или C:\)
.parts      Части пути как кортеж

Пример для /home/user/photo.png:
    p.parent  -> /home/user
    p.name    -> photo.png
    p.stem    -> photo
    p.suffix  -> .png

--------------------------------------------------------------------
3. ПРОВЕРКИ (возвращают True / False)
--------------------------------------------------------------------
.exists()              Файл/папка существует?
.is_file()             Это файл?
.is_dir()              Это папка?
.is_absolute()         Путь абсолютный? (начинается с /)
.is_relative_to(p)     Находится внутри папки p?

--------------------------------------------------------------------
4. ПОСТРОЕНИЕ ПУТЕЙ
--------------------------------------------------------------------
BASE / "sub" / "file.txt"     Объединить пути
p.with_name("new.png")        Заменить имя
p.with_suffix(".jpg")         Заменить расширение
p.with_stem("new_name")       Заменить имя без расширения
p.joinpath("a", "b")          Добавить части
p.resolve()                   Абсолютный путь
Path.cwd()                    Текущая папка
Path.home()                   Домашняя папка

--------------------------------------------------------------------
5. РАБОТА С ПАПКАМИ
--------------------------------------------------------------------
.mkdir()                                 Создать папку
.mkdir(parents=True, exist_ok=True)      Создать безопасно
.iterdir()                               Перебрать содержимое
.glob("*.png")                           Найти по маске
.rglob("*.png")                          Рекурсивный поиск
.rmdir()                                 Удалить пустую папку

--------------------------------------------------------------------
6. РАБОТА С ФАЙЛАМИ
--------------------------------------------------------------------
.open()                Открыть файл
.read_text()           Прочитать текст
.read_bytes()          Прочитать байты
.write_text("...")     Записать текст
.write_bytes(b"...")   Записать байты
.unlink()              Удалить файл
.rename("new.txt")     Переименовать
.stat().st_size        Размер в байтах

--------------------------------------------------------------------
7. ГОТОВЫЕ ПАТТЕРНЫ
--------------------------------------------------------------------

# Базовая настройка
BASE_DIR = Path(__file__).parent
INPUT = BASE_DIR / "data" / "input.png"
OUTPUT = BASE_DIR / "data" / "output.png"

# Безопасное сохранение (создать папку если нет)
OUTPUT.parent.mkdir(parents=True, exist_ok=True)
img.save(OUTPUT)

# Проверка перед чтением
if INPUT.exists() and INPUT.is_file():
    text = INPUT.read_text()

# Найти все PNG в папке
for png in BASE_DIR.glob("*.png"):
    print(png.name)

# Перебрать все файлы
for item in BASE_DIR.iterdir():
    if item.is_file():
        print(item.name)

# Переименовать все PNG в JPG
for f in BASE_DIR.glob("*.png"):
    f.rename(f.with_suffix(".jpg"))

--------------------------------------------------------------------
8. ЧАСТЫЕ ОШИБКИ
--------------------------------------------------------------------
НЕВЕРНО                  ВЕРНО
--------------------     ------------------------------
p.parent()               p.parent         (parent - свойство)
p.mkdir()                p.mkdir(exist_ok=True)
"path" / "file"          Path("path") / "file"
p + "/file"              p / "file"
