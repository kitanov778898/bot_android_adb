-f 0x00020000  — это шестнадцатеричный код флага  FLAG_ACTIVITY_REORDER_TO_FRONT . Он говорит системе: найди уже запущенный экземпляр этого приложения в стеке задач и просто перемести его наверх, не перезагружая его жизненный цикл.

📡 GETEVENT — ЧТЕНИЕ КАСАНИЙ ТАЧСКРИНА (РАБОТАЕТ НА POCO)
═══════════════════════════════════════════════════════════════════════════════

adb shell getevent -lp /dev/input/event3
    # Показать параметры тачскрина
    # Ищи: ABS_MT_POSITION_X (min, max)
    #      ABS_MT_POSITION_Y (min, max)
    # Для POCO: max X=17280, max Y=38400

adb shell getevent -lt /dev/input/event3
    # Показать касания в реальном времени (hex-формат)
    # Остановить: Ctrl+C

adb shell getevent -lt /dev/input/event3 > touch_log.txt
    # Записать касания в файл
    # Потыкай пальцем в экран телефона
    # Потом Ctrl+C — файл готов

import subprocess
import os

AUDIO = os.path.expanduser("~/storage/shared/Music")

def say(text, lang="ru"):
    subprocess.run(["termux-tts-speak", "-l", lang, text])

def play(filename):
    path = os.path.join(AUDIO, filename)
    subprocess.Popen(["termux-media-player", "play", path],
                     stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)

def record(name="rec.wav", sec=5):
    path = os.path.join(AUDIO, name)
    subprocess.run(["termux-microphone-record",
                    "-f", path, "-e", "wav", "-l", str(sec)])
    return path

def stop_all():
    subprocess.run(["termux-microphone-record", "-q"], capture_output=True)
    subprocess.run(["termux-media-player", "stop"], capture_output=True)
    subprocess.run(["pkill", "-f", "play-audio"], capture_output=True)

if __name__ == "__main__":
    say("Начинаю запись")
    rec = record("test.wav", 3)
    say("Запись завершена")
    play("test.wav")
